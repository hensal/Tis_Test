facilities一覧取得GETparent指定
================================

Purpose:
Postman steps for Excel cases 1 through 8:
GET /facilities?parentId=<facility_id>

This document uses separate DB data for each role:
- SYS_ADMIN test data starts with 3000000
- USER01 test data starts with 400000

USER01 role used by SQL:
R_USER01

If your USER01 token has a different role ID, replace R_USER01 in the SQL.

SYS_ADMIN Response Check SQL
----------------------------

Replace <parent_id> with the SYS parent ID for each case.

SELECT
  ft.facility_id,
  ft.parent_id,
  ft.facility_name,
  ft.public_flag AS publish_mode,
  ft.outer_flag AS is_outdoor,
  EXISTS (
    SELECT 1 FROM facility_images fi
    WHERE fi.facility_id = ft.facility_id
      AND fi.deleted_at IS NULL
  ) AS has_vr_image,
  EXISTS (
    SELECT 1 FROM annotations a
    WHERE a.facility_id = ft.facility_id
      AND a.annotation_type = '10'
      AND a.deleted_at IS NULL
  ) AS has_emergency_annotation,
  EXISTS (
    SELECT 1 FROM annotations a
    WHERE a.facility_id = ft.facility_id
      AND a.annotation_type <> '10'
      AND a.deleted_at IS NULL
  ) AS has_normal_annotation,
  EXISTS (
    SELECT 1 FROM equipments e
    WHERE e.facility_id = ft.facility_id
      AND e.deleted_at IS NULL
  ) AS has_equipment,
  ft.created_at,
  ft.updated_at,
  ft.deleted_at
FROM facilities_tree ft
WHERE ft.parent_id = <parent_id>
  AND ft.deleted_at IS NULL
ORDER BY ft.sort_order ASC, ft.facility_id ASC;


USER01 Response Check SQL
-------------------------

Replace <parent_id> with the USER parent ID for each case.

SELECT DISTINCT
  ft.facility_id,
  ft.parent_id,
  ft.facility_name,
  ft.public_flag AS publish_mode,
  ft.outer_flag AS is_outdoor,
  (
    EXISTS (
      SELECT 1 FROM facility_images fi
      WHERE fi.facility_id = ft.facility_id
        AND fi.deleted_at IS NULL
    )
    AND EXISTS (
      SELECT 1 FROM facility_role_permission frp_vr
      WHERE frp_vr.facility_id = ft.facility_id
        AND frp_vr.permission_id = 20
        AND frp_vr.keycloak_role_id IN ('R_USER01')
        AND frp_vr.deleted_at IS NULL
    )
  ) AS has_vr_image,
  (
    EXISTS (
      SELECT 1 FROM annotations a
      WHERE a.facility_id = ft.facility_id
        AND a.annotation_type = '10'
        AND a.deleted_at IS NULL
    )
    AND EXISTS (
      SELECT 1 FROM facility_role_permission frp_ann
      WHERE frp_ann.facility_id = ft.facility_id
        AND frp_ann.permission_id IN (30, 40)
        AND frp_ann.keycloak_role_id IN ('R_USER01')
        AND frp_ann.deleted_at IS NULL
    )
  ) AS has_emergency_annotation,
  (
    EXISTS (
      SELECT 1 FROM annotations a
      WHERE a.facility_id = ft.facility_id
        AND a.annotation_type <> '10'
        AND a.deleted_at IS NULL
    )
    AND EXISTS (
      SELECT 1 FROM facility_role_permission frp_ann
      WHERE frp_ann.facility_id = ft.facility_id
        AND frp_ann.permission_id IN (30, 40)
        AND frp_ann.keycloak_role_id IN ('R_USER01')
        AND frp_ann.deleted_at IS NULL
    )
  ) AS has_normal_annotation,
  (
    EXISTS (
      SELECT 1 FROM equipments e
      WHERE e.facility_id = ft.facility_id
        AND e.deleted_at IS NULL
    )
    AND EXISTS (
      SELECT 1 FROM facility_role_permission frp_eq
      WHERE frp_eq.facility_id = ft.facility_id
        AND frp_eq.permission_id = 50
        AND frp_eq.keycloak_role_id IN ('R_USER01')
        AND frp_eq.deleted_at IS NULL
    )
  ) AS has_equipment,
  ft.created_at,
  ft.updated_at,
  ft.deleted_at
FROM facilities_tree ft
JOIN facility_role_permission frp
  ON frp.facility_id = ft.facility_id
 AND frp.keycloak_role_id IN ('R_USER01')
 AND frp.deleted_at IS NULL
WHERE ft.parent_id = <parent_id>
  AND ft.deleted_at IS NULL
ORDER BY ft.sort_order ASC, ft.facility_id ASC;
